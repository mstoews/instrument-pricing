cmake_minimum_required(VERSION 3.16)
project(pricing_core VERSION 0.1.0 LANGUAGES CXX)
add_library(pricing_core src/black_scholes.cpp)
target_include_directories(pricing_core PUBLIC include)
target_compile_features(pricing_core PUBLIC cxx_std_17)
target_compile_options(pricing_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(pricing_tests tests/black_scholes_test.cpp)
 target_link_libraries(pricing_tests PRIVATE pricing_core)
 add_test(NAME black_scholes COMMAND pricing_tests)
endif()
